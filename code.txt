player-adaptive.js — Lecteur HLS avec ABR custom + override manuel

```javascript
/**
 * player-adaptive.js
 * Lecteur vidéo HLS avec :
 *  - Détection automatique de la bande passante
 *  - Sélection du niveau selon les seuils demandés :
 *      ≤ 5 Mbps   → 144p
 *      5–10 Mbps  → 240p
 *      10–20 Mbps → 360p
 *      20–30 Mbps → 480p
 *      > 30 Mbps  → 720p
 *  - Override manuel (pixels) avec persistance localStorage
 */

export const QUALITY_THRESHOLDS = [
  { name: "144p", height: 144, maxMbps: 5 },
  { name: "240p", height: 240, maxMbps: 10 },
  { name: "360p", height: 360, maxMbps: 20 },
  { name: "480p", height: 480, maxMbps: 30 },
  { name: "720p", height: 720, maxMbps: Infinity },
];

/** Renvoie le niveau recommandé pour un débit donné (Mbps). */
export function getQualityForBandwidth(mbps) {
  for (const q of QUALITY_THRESHOLDS) {
    if (mbps <= q.maxMbps) return q;
  }
  return QUALITY_THRESHOLDS[QUALITY_THRESHOLDS.length - 1];
}

export class AdaptiveVideoPlayer {
  /**
   * @param {HTMLVideoElement} videoEl
   * @param {string} hlsUrl URL du master.m3u8
   * @param {object} opts
   * @param {string} [opts.overrideKey] clé localStorage (défaut: "hls-quality-override")
   * @param {HTMLElement} [opts.selectEl] <select> à peupler avec les qualités
   * @param {(height:number|"auto")=>void} [opts.onChange]
   */
  constructor(videoEl, hlsUrl, opts = {}) {
    this.video = videoEl;
    this.hlsUrl = hlsUrl;
    this.overrideKey = opts.overrideKey || "hls-quality-override";
    this.selectEl = opts.selectEl || null;
    this.onChange = opts.onChange || (() => {});
    this.hls = null;
    this.levels = [];
    this.native = false;
  }

  async init() {
    const { default: Hls } = await import("hls.js");

    if (Hls.isSupported()) {
      this.hls = new Hls({ enableWorker: true });
      this.hls.loadSource(this.hlsUrl);
      this.hls.attachMedia(this.video);

      this.hls.on(Hls.Events.MANIFEST_PARSED, () => {
        this.levels = this.hls.levels
          .map((l, i) => ({ index: i, height: l.height, bitrate: l.bitrate }))
          .sort((a, b) => a.height - b.height);
        this.buildSelect();
        this.applyPreference();
        this.listenConnectionChange();
      });
    } else if (this.video.canPlayType("application/vnd.apple.mpegurl")) {
      // Safari / iOS : HLS natif — on ne peut pas changer de niveau manuellement
      this.video.src = this.hlsUrl;
      this.native = true;
      this.buildSelect(true);
      this.onChange("auto");
    } else {
      throw new Error("HLS non supporté par ce navigateur.");
    }
  }

  /** Mesure le débit disponible (Mbps). */
  async measureBandwidthMbps() {
    if (navigator.connection?.downlink) {
      return navigator.connection.downlink; // Chromium
    }
    // Fallback : on télécharge un petit fichier de test
    try {
      const url = new URL("speedtest.bin", this.hlsUrl).toString();
      const t0 = performance.now();
      const res = await fetch(`${url}?t=${Date.now()}`, { cache: "no-store" });
      const blob = await res.blob();
      const sec = (performance.now() - t0) / 1000;
      return (blob.size * 8) / sec / 1_000_000;
    } catch {
      return null;
    }
  }

  async applyPreference() {
    const override = localStorage.getItem(this.overrideKey);
    if (override && override !== "auto") {
      this.setLevelByHeight(parseInt(override, 10), /*persist*/ false);
      return;
    }
    await this.applyAuto();
  }

  async applyAuto() {
    if (this.native) return;
    const mbps = await this.measureBandwidthMbps();
    if (mbps == null) return;
    const target = getQualityForBandwidth(mbps);
    this.setLevelByHeight(target.height, /*persist*/ false);
    console.info(`[ABR] ${mbps.toFixed(1)} Mbps → ${target.name}`);
  }

  setLevelByHeight(height, persist = true) {
    if (this.native) return;

    // Trouve le niveau le plus proche de la hauteur demandée
    let best = this.levels[0];
    for (const l of this.levels) {
      if (Math.abs(l.height - height) < Math.abs(best.height - height)) best = l;
    }
    if (!best) return;

    this.hls.currentLevel = best.index; // désactive l'ABR interne
    if (persist) {
      if (height === "auto") localStorage.removeItem(this.overrideKey);
      else localStorage.setItem(this.overrideKey, String(best.height));
    }
    if (this.selectEl) this.selectEl.value = String(best.height);
    this.onChange(best.height);
  }

  setManualQuality(value) {
    if (value === "auto") {
      localStorage.removeItem(this.overrideKey);
      this.applyAuto();
    } else {
      this.setLevelByHeight(parseInt(value, 10), true);
    }
  }

  buildSelect(native = false) {
    if (!this.selectEl) return;
    this.selectEl.innerHTML = "";
    const auto = document.createElement("option");
    auto.value = "auto";
    auto.textContent = native ? "Auto (navigateur)" : "Auto (recommandé)";
    this.selectEl.appendChild(auto);

    const heights = native
      ? QUALITY_THRESHOLDS.map((q) => q.height)
      : this.levels.map((l) => l.height);

    heights.forEach((h) => {
      const opt = document.createElement("option");
      opt.value = String(h);
      opt.textContent = `${h}p`;
      this.selectEl.appendChild(opt);
    });

    const saved = localStorage.getItem(this.overrideKey);
    this.selectEl.value = saved || "auto";

    this.selectEl.addEventListener("change", (e) => {
      this.setManualQuality(e.target.value);
    });
  }

  listenConnectionChange() {
    const conn = navigator.connection;
    if (!conn?.addEventListener) return;
    conn.addEventListener("change", () => {
      if (!localStorage.getItem(this.overrideKey)) this.applyAuto();
    });
  }
}

 * Constantes & helpers ABR custom (à placer au niveau module,
 * juste en dessous de la définition de QualityLevel)

/**
 * Seuils de sélection automatique de la qualité :
 *   ≤ 5 Mbps   → 144p
 *   5–10 Mbps  → 240p
 *   10–20 Mbps → 360p
 *   20–30 Mbps → 480p
 *   > 30 Mbps  → 720p
 */
const QUALITY_THRESHOLDS: { height: number; maxMbps: number }[] = [
  { height: 144, maxMbps: 5 },
  { height: 240, maxMbps: 10 },
  { height: 360, maxMbps: 20 },
  { height: 480, maxMbps: 30 },
  { height: 720, maxMbps: Infinity },
];

/** Clé localStorage pour la préférence manuelle (pixels). */
const QUALITY_OVERRIDE_KEY = "hls-quality-override";

/** Renvoie la hauteur cible pour un débit donné (Mbps). */
function getTargetHeightForBandwidth(mbps: number): number {
  for (const q of QUALITY_THRESHOLDS) {
    if (mbps <= q.maxMbps) return q.height;
  }
  return 720;
}

/** Lit `navigator.connection.downlink` (Mbps) si disponible. */
function readBandwidthMbps(): number | null {
  const conn =
    (navigator as any).connection ||
    (navigator as any).mozConnection ||
    (navigator as any).webkitConnection;
  if (conn && typeof conn.downlink === "number" && conn.downlink > 0) {
    return conn.downlink;
  }
  return null;
}

/** Trouve l'index de niveau HLS le plus proche de `targetHeight`. */
function findBestLevelIndex(levels: QualityLevel[], targetHeight: number): number {
  if (!levels.length) return -1;
  const sorted = [...levels].sort((a, b) => a.height - b.height);
  let best = sorted[0];
  for (const lvl of sorted) {
    if (lvl.height <= targetHeight && lvl.height >= best.height) best = lvl;
  }
  // Si toutes les pistes sont au-dessus de la cible → la plus petite
  if (best.height > targetHeight) return sorted[0].id;
  return best.id;
}

function HybridVideoPlayer({ src, poster, fallbackSrc }: HybridVideoPlayerProps) {
  const videoRef = useRef<HTMLVideoElement | null>(null);

  const mediaRecoveryAttempts = useRef<number>(0);

  const [playbackError, setPlaybackError] = useState<{ message: string; isNetwork: boolean } | null>(null);
  const [currentLevelResolution, setCurrentLevelResolution] = useState<string | null>(null);
  const [isFallbackActive, setIsFallbackActive] = useState(false);
  const [retryKey, setRetryKey] = useState(0);

  const [currentQualityIndex, setCurrentQualityIndex] = useState<number>(-1); // -1 = Auto
  const [playbackSpeed, setPlaybackSpeed] = useState<number>(1);
  const [showSettingsMenu, setShowSettingsMenu] = useState(false);

  // Débit courant (pour affichage éventuel dans le menu Auto)
  const [bandwidthMbps, setBandwidthMbps] = useState<number | null>(null);

  // Synchronise lorsque l'URL source parent change
  useEffect(() => {
    setActiveSrc(src);
    setIsFallbackActive(false);
    setCurrentQualityIndex(-1);
    setLevels([]);
  }, [src]);

  const streamType = useMemo(() => detectStreamType(activeSrc), [activeSrc]);

  const cleanupHls = useCallback(() => {
    if (hlsRef.current) {
      hlsRef.current.stopLoad();
      hlsRef.current.detachMedia();
      hlsRef.current.destroy();

    const mbps = readBandwidthMbps();

      return;

    const idx = findBestLevelIndex(levels, targetHeight);
    if (idx >= 0) {
      hls.currentLevel = idx;
    }

      const hls = hlsRef.current;
      if (!hls || levels.length === 0) return -1;
      const idx = findBestLevelIndex(levels, targetHeight);
      if (idx >= 0) {
        hls.currentLevel = idx;
        setCurrentQualityIndex(idx);

      cleanupHls();

        const bufferedEnd = buffered.end(i);
        const pct = Math.min(100, Math.round((bufferedEnd / video.duration) * 100));
        setBufferProgress(pct);

  }, []);

  useEffect(() => {

    if (!video || !activeSrc) return;

    setPlaybackError(null);
    setIsBuffering(true);
    setBufferProgress(null);
    setCurrentLevelResolution(null);
    mediaRecoveryAttempts.current = 0;

        setRetryKey((prev) => prev + 1);
      }
    };

    video.addEventListener("waiting", handleWaiting);
    video.addEventListener("stalled", handleStalled);
    video.addEventListener("playing", handlePlaying);
    video.addEventListener("canplay", handleCanPlay);
    video.addEventListener("progress", handleProgress);
    video.addEventListener("error", handleError);
    window.addEventListener("online", handleOnline);

    return () => {
      video.removeEventListener("waiting", handleWaiting);
      video.removeEventListener("stalled", handleStalled);
      video.removeEventListener("playing", handlePlaying);

      (navigator as any).mozConnection ||

      if (stored && stored !== "auto") return;

    setPlaybackError(null);
    setIsBuffering(true);
    setRetryKey((prev) => prev + 1);
  };

  // ---- Changement de qualité : Auto ↔ pixels (avec persistance) ---------
  const changeQuality = (levelIndex: number) => {
    setShowSettingsMenu(false);
    if (!hlsRef.current) return;

    if (levelIndex === -1) {
      // Retour en Auto piloté par les seuils
      try { localStorage.removeItem(QUALITY_OVERRIDE_KEY); } catch { /* ignore */ }
      setCurrentQualityIndex(-1);
      applyAutoQuality();
      toast.info("Qualité : Automatique (selon débit)");
    } else {
      // Sélection manuelle (pixels)
      hlsRef.current.currentLevel = levelIndex;
      setCurrentQualityIndex(levelIndex);
      const selected = levels[levelIndex];
      if (selected) {
        try {
          localStorage.setItem(QUALITY_OVERRIDE_KEY, String(selected.height));
        } catch { /* ignore */ }
        toast.info(`Qualité : ${selected.label}`);
      }
    }
  };

  const changeSpeed = (speed: number) => {
    setPlaybackSpeed(speed);
    if (videoRef.current) videoRef.current.playbackRate = speed;
    setShowSettingsMenu(false);
    toast.info(`Vitesse : ${speed}x`);
  };

  // Résolution cible affichée dans le menu Auto
  const autoTargetLabel = useMemo(() => {
    if (bandwidthMbps == null) return null;
    const h = getTargetHeightForBandwidth(bandwidthMbps);
    return `${h}p`;
  }, [bandwidthMbps]);

  return (
    <div className="relative w-full h-full bg-black flex items-center justify-center overflow-hidden group select-none">
      <video
        ref={videoRef}
        poster={poster}
        controls
        preload="metadata"
        playsInline
        className="w-full h-full object-contain"
      >
        {streamType === "mp4" && <source src={activeSrc} type="video/mp4" />}
      </video>

      {/* Badges d'information */}
      <div className="absolute top-3 left-3 z-20 pointer-events-none opacity-80 group-hover:opacity-100 transition-opacity flex items-center gap-1.5">
        <span className="text-[10px] font-semibold tracking-wider uppercase px-2 py-0.5 rounded bg-black/60 backdrop-blur-sm text-zinc-300 border border-white/10">
          {isFallbackActive
            ? "MP4 (Secours)"
            : streamType === "mp4"
            ? "MP4 Standard"
            : "HLS Adaptatif"}
        </span>
        {streamType === "hls" && currentLevelResolution && (
          <span className="text-[10px] font-bold px-1.5 py-0.5 rounded bg-primary/80 backdrop-blur-sm text-white">
            {currentLevelResolution}
          </span>
        )}
      </div>

      {/* Paramètres de lecture */}
      <div className="absolute top-3 right-3 z-20">
        <button
          type="button"
          onClick={() => setShowSettingsMenu(!showSettingsMenu)}
          aria-label="Paramètres de lecture"
          className="p-1.5 rounded-full bg-black/60 hover:bg-black/80 backdrop-blur-sm text-white/80 hover:text-white border border-white/10 transition-all shadow-md"
        >
          <Settings className="h-4 w-4" />
        </button>

        {showSettingsMenu && (
          <div className="absolute right-0 mt-2 w-52 rounded-xl bg-zinc-900/95 backdrop-blur-md border border-white/15 p-2 shadow-2xl z-30 text-xs text-white">
            {streamType === "hls" && levels.length > 0 && (
              <div className="mb-2">
                <div className="font-semibold text-zinc-400 uppercase text-[10px] px-2 py-1">
                  Résolution (pixels)
                </div>

                {/* Option Auto : affiche la cible calculée */}
                <button
                  type="button"
                  onClick={() => changeQuality(-1)}
                  className={`w-full text-left px-2 py-1.5 rounded flex items-center justify-between hover:bg-white/10 ${
                    currentQualityIndex === -1 ? "text-primary font-bold" : "text-zinc-200"
                  }`}
                >
                  <span>
                    Auto
                    {autoTargetLabel && (
                      <span className="ml-1 text-[10px] text-zinc-400">
                        ({autoTargetLabel})
                      </span>
                    )}
                  </span>
                  {currentQualityIndex === -1 && <Check className="h-3 w-3" />}
                </button>

                {levels
                  .slice()
                  .sort((a, b) => a.height - b.height)
                  .map((lvl) => (
                    <button
                      key={lvl.id}
                      type="button"
                      onClick={() => changeQuality(lvl.id)}
                      className={`w-full text-left px-2 py-1.5 rounded flex items-center justify-between hover:bg-white/10 ${
                        currentQualityIndex === lvl.id
                          ? "text-primary font-bold"
                          : "text-zinc-200"
                      }`}
                    >
                      <span>{lvl.label}</span>
                      {currentQualityIndex === lvl.id && <Check className="h-3 w-3" />}
                    </button>
                  ))}

                <div className="my-1 border-t border-white/10" />
              </div>
            )}

            <div>
              <div className="font-semibold text-zinc-400 uppercase text-[10px] px-2 py-1">
                Vitesse
              </div>
              {[0.75, 1, 1.25, 1.5, 2].map((s) => (
                <button
                  key={s}
                  type="button"
                  onClick={() => changeSpeed(s)}
                  className={`w-full text-left px-2 py-1.5 rounded flex items-center justify-between hover:bg-white/10 ${
                    playbackSpeed === s ? "text-primary font-bold" : "text-zinc-200"
                  }`}
                >
                  <span>{s === 1 ? "Normal (1x)" : `${s}x`}</span>
                  {playbackSpeed === s && <Check className="h-3 w-3" />}
                </button>
              ))}
            </div>
          </div>
        )}
      </div>

      {isBuffering && !playbackError && (
        <BufferingLoader progressPct={bufferProgress} />
      )}

      {playbackError && (
        <VideoErrorOverlay
          message={playbackError.message}
          isNetworkError={playbackError.isNetwork}
          onRetry={handleRetry}
        />
      )}
    </div>
  );
}
